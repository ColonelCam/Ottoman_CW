	}
	mean_time_to_happen = { days = 95 }
	option = {
		name = merf.1.a
		add_research_slot = 1
		FSW = {
			SWE_sven_olov_knutsson_lindholm = {
				set_nationality = DEN
			}
			transfer_units_fraction= {
				target = ROOT
				size = 1.0
				stockpile_ratio = 1.0
				army_ratio = 1.0
				navy_ratio = 1.0
				air_ratio = 1.0
				keep_unit_leaders_trigger = {
					always = no
				}
			}
		}
		every_state = {
			limit = {
				is_core_of = FSW
			}
			add_core_of = DEN
		}
		annex_country = {
		    target = FSW
		    transfer_troops = yes
		}
		add_country_leader_role = {
		    character = SWE_sven_olov_knutsson_lindholm
		    promote_leader = yes
		    country_leader = {
		        ideology = nazism
		        traits = { dictator }
		    }
		}
		news_event = {
			id = merf.2
			hours = 10
		}
		country_event = {
			id = merf.3
			days = 250
		}
		set_cosmetic_tag = SCA_UNIFIED
		set_global_flag = form_scandinavia_flag
		set_party_name = {
		    ideology = fascism
		    long_name = FSW_fascism_party_long
		    name = FSW_fascism_party
		}
	}
}
